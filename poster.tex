\copyblock{9}{45}{90}{\begin{itemize}
\item University information is distributed across different sources (e.g., CampusPortal, wikis and manuals).
\item Finding relevant and up-to-date answers is time-consuming.
\item Traditional search can struggle with natural-language questions and process-related context.
\item University information requires privacy-conscious processing.
\end{itemize}}
\copyblock{111}{45}{90}{\fontsize{7.8}{9}\selectfont\begin{itemize}\setlength{\itemsep}{.9mm}
\item Enable \textbf{natural-language question answering} about university processes.
\item Retrieve relevant information from \textbf{approved knowledge sources}.
\item Generate context-aware answers using RAG.
\item Provide answers with source references and URLs.
\item Support a \textbf{locally hosted solution} without external LLM APIs.
\item Evaluate retrieval quality, correctness and response time.
\end{itemize}}
% Methodology
\panel{5}{89}{200}{57}{\textcircled{\small *}\quad Methodology}
\draw[line,line width=.4pt] (105,102) -- (105,142);
\copyblock{9}{102}{91}{
\step{Data Ingestion}{Available university knowledge records are collected from CSV files, combining synthetic data and publicly available THWS information.}
\step{Data Preprocessing}{Questions, answers, key facts, evidence and source details are cleaned and prepared for further processing.}
\step{Text Chunking}{Longer records are divided into smaller text passages while preserving relevant metadata and source information.}}
\copyblock{110}{102}{91}{
\step{Embedding Generation}{BGE-M3 converts text passages into numerical vector representations for semantic search.}
\step{Information Retrieval}{ChromaDB stores embeddings and retrieves relevant passages based on the user's natural-language query.}
\step{Answer Generation}{Qwen 2.5 7B Instruct, running locally through Ollama, generates context-aware answers using retrieved information and provides supporting source references.}}
